Q:
Print GFG n times

Sol: (C#)
class Solution
{
    //Complete this function
    //Function to print the value of N for GfG.
    public void printGfg(int N)
    {
        //Your code here
        if(N < 1){
            return;
        }
        Console.Write("GFG ");
        printGfg(N-1);
    }
}
