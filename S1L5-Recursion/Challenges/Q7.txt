Q:
125. Valid Palindrome

Sol: (Normal approach)
public class Solution {
    public bool IsPalindrome(string s) {
        s = new String(s.Where(c => Char.IsLetter(c) || Char.IsNumber(c)).ToArray()).ToLower();

        int start = 0;
        int end = s.Length-1;

        while(start<end){
            if(s[start] != s[end]){
                return false;
            }
            start++;
            end--;
        }
        return true;
    }
}
