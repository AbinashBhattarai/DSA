Q:
Print N to 1 without loop

Sol: (C#)
class Solution
{
    //Complete this function
    //Function to print numbers from 1 to N.
    public void printNos(int N)
    {
        //Your code here
        if(N < 1){
            return;
        }
        Console.Write(N + " ");
        printNos(N - 1);
    }
}
